    <script>
        // Feriados Nacionais Brasileiros de 2026
        const nationalHolidays2026 = [
            '2026-01-01', // Ano Novo
            '2026-02-16', // Segunda-feira de Carnaval
            '2026-02-17', // Carnaval
            '2026-02-18', // Quarta-feira de Cinzas
            '2026-04-21', // Tiradentes
            '2026-05-01', // Dia do Trabalho
            '2026-09-07', // Independência
            '2026-10-12', // Nossa Senhora Aparecida
            '2026-11-02', // Finados
            '2026-11-20', // Consciência Negra
            '2026-12-25', // Natal
        ];

        // Estados dos elementos
        let currentStep = 1;
        let bookingData = {
            date: '',
            startTime: '',
            endTime: '',
            gasService: false,
            fullName: '',
            phone: '',
            email: '',
            total: 500,
        };

        // Elementos do DOM
        const rulesCheckbox = document.getElementById('rulesCheckbox');
        const continueBtn = document.getElementById('continueBtn');
        const reservationDateInput = document.getElementById('reservationDate');
        const startTimeInput = document.getElementById('startTime');
        const endTimeInput = document.getElementById('endTime');
        const gasServiceCheckbox = document.getElementById('gasService');
        const fullNameInput = document.getElementById('fullName');
        const phoneInput = document.getElementById('phone');
        const emailInput = document.getElementById('email');
        const bookingForm = document.getElementById('bookingForm');
        const successSection = document.getElementById('successSection');

        // Event Listeners
        rulesCheckbox.addEventListener('change', toggleContinueButton);
        reservationDateInput.addEventListener('change', validateDate);
        startTimeInput.addEventListener('change', calculateEndTime);
        gasServiceCheckbox.addEventListener('change', updateTotal);
        phoneInput.addEventListener('input', formatPhoneNumber);
        bookingForm.addEventListener('submit', handleFormSubmit);

        // Habilitar/Desabilitar botão de continuar
        function toggleContinueButton() {
            continueBtn.disabled = !rulesCheckbox.checked;
        }

        // Validar se a data é fim de semana ou feriado
        function isValidReservationDate(dateString) {
            const date = new Date(dateString + 'T00:00:00');
            const dayOfWeek = date.getDay();
            
            // Verificar se é fim de semana (sábado = 6, domingo = 0)
            const isWeekend = dayOfWeek === 0 || dayOfWeek === 6;
            
            // Verificar se é feriado nacional
            const isHoliday = nationalHolidays2026.includes(dateString);
            
            return isWeekend || isHoliday;
        }

        // Validar data
        function validateDate() {
            const selectedDate = reservationDateInput.value;
            const dateError = document.getElementById('dateError');
            
            if (!selectedDate) {
                dateError.classList.remove('show');
                return;
            }
            
            // Verificar se é data futura
            const today = new Date();
            today.setHours(0, 0, 0, 0);
            const selected = new Date(selectedDate + 'T00:00:00');
            
            if (selected < today) {
                dateError.textContent = '❌ A data não pode ser no passado.';
                dateError.classList.add('show');
                reservationDateInput.value = '';
                return;
            }
            
            // Verificar se é fim de semana ou feriado
            if (!isValidReservationDate(selectedDate)) {
                dateError.textContent = '❌ Reservas permitidas apenas nos finais de semana e feriados nacionais.';
                dateError.classList.add('show');
                reservationDateInput.value = '';
                return;
            }
            
            dateError.classList.remove('show');
            bookingData.date = selectedDate;
            updateReviewDate();
        }

        // Calcular horário de término (10 horas de aluguel)
        function calculateEndTime() {
            const startTime = startTimeInput.value;
            
            if (!startTime) {
                endTimeInput.value = '';
                return;
            }
            
            // Extrair horas e minutos
            const [hours, minutes] = startTime.split(':').map(Number);
            
            // Adicionar 10 horas
            let endHours = hours + 10;
            let endMinutes = minutes;
            
            // Ajustar se passar de 24 horas
            if (endHours >= 24) {
                endHours -= 24;
            }
            
            // Formatar a hora
            const endTime = `${String(endHours).padStart(2, '0')}:${String(endMinutes).padStart(2, '0')}`;
            endTimeInput.value = endTime;
            
            bookingData.startTime = startTime;
            bookingData.endTime = endTime;
            updateReviewTime();
        }

        // Atualizar total quando selecionar gás
        function updateTotal() {
            bookingData.gasService = gasServiceCheckbox.checked;
            const basePrice = 500;
            bookingData.total = bookingData.gasService ? basePrice + 35 : basePrice;
            updateReviewPrice();
        }

        // Formatar número de telefone
        function formatPhoneNumber(e) {
            let value = e.target.value.replace(/\D/g, '');
            
            if (value.length > 11) {
                value = value.slice(0, 11);
            }
            
            if (value.length <= 2) {
                value = value;
            } else if (value.length <= 6) {
                value = `(${value.slice(0, 2)}) ${value.slice(2)}`;
            } else if (value.length <= 10) {
                value = `(${value.slice(0, 2)}) ${value.slice(2, 6)}-${value.slice(6)}`;
            } else {
                value = `(${value.slice(0, 2)}) ${value.slice(2, 7)}-${value.slice(7)}`;
            }
            
            e.target.value = value;
        }

        // Validar nome
        function validateName(name) {
            const nameRegex = /^[a-záéíóúâêôãõç\s]{3,}$/i;
            return nameRegex.test(name);
        }

        // Validar telefone
        function validatePhone(phone) {
            const phoneRegex = /^\(\d{2}\) \d{4,5}-\d{4}$/;
            return phoneRegex.test(phone);
        }

        // Validar email (opcional)
        function validateEmail(email) {
            if (!email) return true; // Email é opcional
            const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            return emailRegex.test(email);
        }

        // Atualizar revisão de data
        function updateReviewDate() {
            const dateSpan = document.getElementById('reviewDate');
            if (bookingData.date) {
                const date = new Date(bookingData.date + 'T00:00:00');
                const options = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' };
                const formattedDate = date.toLocaleDateString('pt-BR', options);
                dateSpan.textContent = formattedDate;
            }
        }

        // Atualizar revisão de horário
        function updateReviewTime() {
            const timeSpan = document.getElementById('reviewTime');
            if (bookingData.startTime && bookingData.endTime) {
                timeSpan.textContent = `${bookingData.startTime} - ${bookingData.endTime} (10 horas)`;
            }
        }

        // Atualizar revisão de preço
        function updateReviewPrice() {
            const gasItem = document.getElementById('gasItem');
            const totalPrice = document.getElementById('totalPrice');
            
            if (bookingData.gasService) {
                gasItem.style.display = 'flex';
            } else {
                gasItem.style.display = 'none';
            }
            
            totalPrice.textContent = `R$ ${bookingData.total.toFixed(2).replace('.', ',')}`;
        }

        // Navegar entre etapas
        function goToStep(stepNumber) {
            // Validar etapa atual antes de avançar
            if (stepNumber > currentStep) {
                if (!validateCurrentStep()) {
                    return;
                }
            }
            
            // Ocultar etapa atual
            document.getElementById(`step${currentStep}`).classList.add('hidden');
            
            // Mostrar nova etapa
            currentStep = stepNumber;
            document.getElementById(`step${currentStep}`).classList.remove('hidden');
            
            // Atualizar revisão
            if (stepNumber === 4) {
                updateReview();
            }
            
            // Rolar para o topo
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Validar etapa atual
        function validateCurrentStep() {
            const nameError = document.getElementById('nameError');
            const phoneError = document.getElementById('phoneError');
            const dateError = document.getElementById('dateError');
            
            switch (currentStep) {
                case 2:
                    // Validar data e horário
                    if (!reservationDateInput.value) {
                        dateError.textContent = '❌ Selecione uma data válida.';
                        dateError.classList.add('show');
                        return false;
                    }
                    
                    if (!startTimeInput.value) {
                        alert('❌ Selecione um horário de início.');
                        return false;
                    }
                    
                    return true;
                    
                case 3:
                    // Validar dados pessoais
                    const fullName = fullNameInput.value.trim();
                    const phone = phoneInput.value.trim();
                    const email = emailInput.value.trim();
                    
                    if (!fullName) {
                        nameError.textContent = '❌ Nome completo é obrigatório.';
                        nameError.classList.add('show');
                        return false;
                    }
                    
                    if (!validateName(fullName)) {
                        nameError.textContent = '❌ Nome deve ter no mínimo 3 caracteres.';
                        nameError.classList.add('show');
                        return false;
                    }
                    nameError.classList.remove('show');
                    
                    if (!phone) {
                        phoneError.textContent = '❌ Telefone é obrigatório.';
                        phoneError.classList.add('show');
                        return false;
                    }
                    
                    if (!validatePhone(phone)) {
                        phoneError.textContent = '❌ Formato de telefone inválido.';
                        phoneError.classList.add('show');
                        return false;
                    }
                    phoneError.classList.remove('show');
                    
                    if (!validateEmail(email)) {
                        alert('❌ Formato de email inválido.');
                        return false;
                    }
                    
                    bookingData.fullName = fullName;
                    bookingData.phone = phone;
                    bookingData.email = email;
                    
                    return true;
                    
                default:
                    return true;
            }
        }

        // Atualizar revisão
        function updateReview() {
            document.getElementById('reviewDate').textContent = formatDateForDisplay(bookingData.date);
            document.getElementById('reviewTime').textContent = `${bookingData.startTime} - ${bookingData.endTime} (10 horas)`;
            document.getElementById('reviewName').textContent = bookingData.fullName;
            document.getElementById('reviewPhone').textContent = bookingData.phone;
            
            if (bookingData.email) {
                document.getElementById('reviewEmailSection').style.display = 'block';
                document.getElementById('reviewEmail').textContent = bookingData.email;
            } else {
                document.getElementById('reviewEmailSection').style.display = 'none';
            }
            
            // Atualizar preço
            const gasItem = document.getElementById('gasItem');
            const totalPrice = document.getElementById('totalPrice');
            
            if (bookingData.gasService) {
                gasItem.style.display = 'flex';
            } else {
                gasItem.style.display = 'none';
            }
            
            totalPrice.textContent = `R$ ${bookingData.total.toFixed(2).replace('.', ',')}`;
        }

        // Formatar data para exibição
        function formatDateForDisplay(dateString) {
            const date = new Date(dateString + 'T00:00:00');
            const options = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' };
            return date.toLocaleDateString('pt-BR', options);
        }

        // Gerar ID único para reserva
        function generateBookingId() {
            return 'RES-' + Date.now() + '-' + Math.random().toString(36).substr(2, 9).toUpperCase();
        }

        // Processar envio do formulário
        function handleFormSubmit(e) {
            e.preventDefault();
            
            // Validar checkbox de confirmação
            const confirmCheckbox = document.getElementById('confirmCheckbox');
            if (!confirmCheckbox.checked) {
                alert('❌ Você deve confirmar que deseja fazer a reserva.');
                return;
            }
            
            // Gerar ID de reserva
            const bookingId = generateBookingId();
            
            // Ocultar formulário
            bookingForm.classList.add('hidden');
            
            // Mostrar tela de sucesso
            successSection.classList.remove('hidden');
            
            // Preencher dados de sucesso
            document.getElementById('bookingId').textContent = bookingId;
            document.getElementById('successDate').textContent = formatDateForDisplay(bookingData.date);
            document.getElementById('successTime').textContent = `${bookingData.startTime} - ${bookingData.endTime}`;
            document.getElementById('successTotal').textContent = `R$ ${bookingData.total.toFixed(2).replace('.', ',')}`;
            
            // Simular envio de dados (em produção, enviar para um servidor)
            console.log('Reserva confirmada:', {
                bookingId,
                ...bookingData,
                confirmadoEm: new Date().toLocaleString('pt-BR'),
            });
            
            // Rolar para o topo
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function setMinimumDate() {
            const today = new Date();
            const year = today.getFullYear();
            const month = String(today.getMonth() + 1).padStart(2, '0');
            const day = String(today.getDate()).padStart(2, '0');
